```csharp
using System;
using System.IO;
using System.Net.Http;
using System.Net.Http.Json;
using System.Threading.Tasks;

var apiKey = Environment.GetEnvironmentVariable("MISTRAL_API_KEY") ?? throw new InvalidOperationException("MISTRAL_API_KEY not set");
var model = Environment.GetEnvironmentVariable("MISTRAL_MODEL") ?? "mistral-medium-latest";
var systemMessage = await File.ReadAllTextAsync("system.md");

var client = new HttpClient
{
    BaseAddress = new Uri("https://api.mistral.ai/v1/"),
    DefaultRequestHeaders = { { "Authorization", $"Bearer {apiKey}" } }
};

Console.WriteLine("Chat with Mistral (press Enter on empty line to exit)");
while (true)
{
    Console.Write("> ");
    var userInput = Console.ReadLine();
    if (string.IsNullOrWhiteSpace(userInput)) break;

    var response = await client.PostAsJsonAsync("chat/completions", new
    {
        model,
        messages = new[]
        {
            new { role = "system", content = systemMessage },
            new { role = "user", content = userInput }
        }
    });

    response.EnsureSuccessStatusCode();
    var result = await response.Content.ReadFromJsonAsync<ChatCompletionResponse>();

    Console.WriteLine($"{result?.choices?[0]?.message?.content}\n");
}

public record ChatCompletionResponse(
    string id,
    string model,
    Choice[] choices,
    Usage usage);

public record Choice(
    int index,
    Message message,
    string? finishReason);

public record Message(
    string role,
    string content);

public record Usage(
    int promptTokens,
    int completionTokens,
    int totalTokens);
```

5. Includes all required response DTOs with correct casing
6. Uses C# 14 features (file-scoped namespace, records, etc.)
