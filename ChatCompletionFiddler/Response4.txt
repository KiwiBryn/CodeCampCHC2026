User input processed in loop,that exits when a blank line is entered. Load Chat Completion API System Message from file system.md. APIKey Environment variable MISTRAL_API_KEY must be set. Model can be set in code, or via environment variable MISTRAL_MODEL. Default model is mistral-medium-latest. Use AddHttpResilience
Here's a concise .NET 10 C#14 console application that meets your requirements:

```csharp

using System;
using System.Net.Http;
using System.Net.Http.Json;
using System.Text.Json.Serialization;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Http.Resilience;
using Polly;

var builder = new HttpClientBuilder()
    .AddResiliencePipeline("chat", pipelineBuilder =>
    {
       pipelineBuilder.AddRetry(new() { MaxRetryAttempts = 3 });
       pipelineBuilder.AddTimeout(TimeSpan.FromSeconds(30));
    });

var httpClient = builder.Build();

var apiKey = Environment.GetEnvironmentVariable("MISTRAL_API_KEY")
    ?? throw new InvalidOperationException("MISTRAL_API_KEY not set");

var model = Environment.GetEnvironmentVariable("MISTRAL_MODEL") ?? "mistral-medium-latest";
var systemMessage = await File.ReadAllTextAsync("system.md");

var request = new
{
   model,
   messages = new object[]
    {
        new { role = "system", content = systemMessage }
    },
   temperature = 0.7
};

while (true)
{
   Console.Write("> ");
   var userInput = Console.ReadLine();

   if (string.IsNullOrWhiteSpace(userInput))
      break;

   var userMessage = new { role = "user", content = userInput };

   request.messages = [.. request.messages, userMessage];

   var response = await httpClient.PostAsJsonAsync(
       "https://api.mistral.ai/v1/chat/completions",
       request,
       new(request) { DefaultIgnoreCondition = JsonIgnoreCondition.WhenWritingNull }
   );

   response.EnsureSuccessStatusCode();

   var result = await response.Content.ReadFromJsonAsync<ChatCompletionResponse>()
       ?? throw new InvalidOperationException("Invalid response");

   Console.WriteLine(result.Choices[0].Message.Content);

   request.messages = request.messages.Append(new { role = "assistant", content = result.Choices[0].Message.Content }).ToArray();
}

public record ChatCompletionResponse(
    string Id,
    string Model,
    long Created,
    Choice[] Choices,
    Usage Usage
);

public record Choice(
    int Index,
    Message Message,
    string? FinishReason
);

public record Message(
    string Role,
    string Content
);

public record Usage(
    int PromptTokens,
    int CompletionTokens,
    int TotalTokens
);
```
