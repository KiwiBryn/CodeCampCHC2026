### Services/MistralAiService.cs
```csharp
using System.Net.Http.Headers;
using System.Net.Http.Json;
using System.Text.Json;
using MistralAi.Agents.Models;

namespace MistralAi.Agents.Services;

public class MistralAiService : IMistralAiService
{
    private readonly HttpClient _httpClient;
    private readonly JsonSerializerOptions _jsonOptions;

    public MistralAiService(HttpClient httpClient)
    {
        _httpClient = httpClient;
        _jsonOptions = new JsonSerializerOptions
        {
            PropertyNamingPolicy = JsonNamingPolicy.SnakeCaseLower,
            DefaultIgnoreCondition = System.Text.Json.Serialization.JsonIgnoreCondition.WhenWritingNull
        };
    }

    public async Task<ChatCompletionResponse> CreateChatCompletionAsync(
        ChatCompletionRequest request,
        CancellationToken cancellationToken = default)
    {
        var response = await _httpClient.PostAsJsonAsync(
            "/chat/completions",
            request,
            _jsonOptions,
            cancellationToken)
            .ConfigureAwait(false);

        response.EnsureSuccessStatusCode();
        return await response.Content.ReadFromJsonAsync<ChatCompletionResponse>(_jsonOptions, cancellationToken)
            .ConfigureAwait(false);
    }

    public async IAsyncEnumerable<ChatCompletionResponse> CreateChatCompletionStreamAsync(
        ChatCompletionRequest request,
        [EnumeratorCancellation] CancellationToken cancellationToken = default)
    {
        request.Stream = true;
        var httpRequest = new HttpRequestMessage(HttpMethod.Post, "/chat/completions")
        {
            Content = JsonContent.Create(request, options: _jsonOptions)
        };

        using var response = await _httpClient.SendAsync(httpRequest, HttpCompletionOption.ResponseHeadersRead, cancellationToken)
            .ConfigureAwait(false);

        response.EnsureSuccessStatusCode();

        using var stream = await response.Content.ReadAsStreamAsync(cancellationToken)
            .ConfigureAwait(false);
        using var reader = new StreamReader(stream);

        while (!reader.EndOfStream && !cancellationToken.IsCancellationRequested)
        {
            var line = await reader.ReadLineAsync(cancellationToken).ConfigureAwait(false);
            if (string.IsNullOrWhiteSpace(line) || line.StartsWith("data: [DONE]"))
                break;

            if (line.StartsWith("data: "))
            {
                var json = line[6..];
                var response = JsonSerializer.Deserialize<ChatCompletionResponse>(json, _jsonOptions);
                if (response != null)
                {
                    yield return response;
                }
            }
        }
    }
}
```

```csharp
namespace SampleConsoleApp;

public class ChatDemo(IMistralAiService mistralService)

    public async Task RunAsync()
    {

            // Basic chat completion example

    private async Task BasicChatCompletionAsync()
    {
        Console.WriteLine("\n=== Basic Chat Completion ===");

        var request = new ChatCompletionRequest
        {
            Model = "mistral-large-latest",
            Messages =
            [
                new ChatMessage { Role = MessageRole.System, Content = "You are a helpful assistant." },
                new ChatMessage { Role = MessageRole.User, Content = "What is the capital of France?" }
            ],
            Temperature = 0.7f,
            MaxTokens = 100
        };

        var response = await mistralService.CreateChatCompletionAsync(request);
        Console.WriteLine($"Response: {response.Choices[0].Message.Content}");
        Console.WriteLine($"Usage: {response.Usage?.TotalTokens} tokens");
    }

    private async Task StreamingChatCompletionAsync()
    {
        Console.WriteLine("\n=== Streaming Chat Completion ===");

        var request = new ChatCompletionRequest
        {
            Model = "mistral-large-latest",
            Messages =
            [
                new ChatMessage { Role = MessageRole.User, Content = "Write a short poem about the ocean." }
            ],
            Temperature = 0.9f,
            MaxTokens = 200
        };

        await foreach (var chunk in mistralService.CreateChatCompletionStreamAsync(request))
        {
            foreach (var choice in chunk.Choices)
            {
                if (choice.Message.Content is string content)
                {
                    Console.Write(content);
                }
            }
        }
        Console.WriteLine();
    }

    private async Task ToolCallingExampleAsync()
    {
        Console.WriteLine("\n=== Tool Calling Example ===");

        var tools = new List<Tool>
        {
            new()
            {
                Type = ToolType.Function,
                Function = new FunctionDefinition
                {
                    Name = "get_weather",
                    Description = "Get the current weather for a location",
                    Parameters = new Dictionary<string, object>
                    {
                        ["type"] = "object",
                        ["properties"] = new Dictionary<string, object>
                        {
                            ["location"] = new Dictionary<string, object>
                            {
                                ["type"] = "string",
                                ["description"] = "The city and state, e.g. San Francisco, CA"
                            }
                        },
                        ["required"] = new[] { "location" }
                    }
                }
            }
        };

        var request = new ChatCompletionRequest
        {
            Model = "mistral-large-latest",
            Messages =
            [
                new ChatMessage
                {
                    Role = MessageRole.User,
                    Content = "What's the weather like in Paris?"
                }
            ],
            Tools = tools,
            ToolChoice = ToolChoice.Auto
        };

        var response = await mistralService.CreateChatCompletionAsync(request);
        if (response.Choices[0].Message.ToolCalls is { Count: > 0 })
        {
            Console.WriteLine("Model requested tool calls:");
            foreach (var toolCall in response.Choices[0].Message.ToolCalls)
            {
                Console.WriteLine($"- {toolCall.Function?.Name}({toolCall.Function?.Arguments})");
            }
        }
        else
        {
            Console.WriteLine($"Response: {response.Choices[0].Message.Content}");
        }
    }
}
```

```csharp
    private async Task InteractiveChatWithToolsAsync(List<Tool> tools)

        Console.WriteLine("\n=== Interactive Chat with File Operations ===");

        var messages = new List<ChatMessage>

            var input = Console.ReadLine();

            if (string.Equals(input, "exit", StringComparison.OrdinalIgnoreCase))

                Console.WriteLine("- Delete file: 'delete file [filename]'");
                Console.WriteLine("- Delete folder: 'delete folder [path]'");

            {

    private async Task ProcessToolCallsAsync(List<ToolCall> toolCalls, List<ChatMessage> messages)
    {
        foreach (var toolCall in toolCalls)
        {
            if (toolCall.Function == null) continue;

            if (_toolFunctions.TryGetValue(toolCall.Function.Name, out var func))
            {
                try
                {
                    // Parse arguments (simplified - in production use proper JSON parsing)
                    var result = await func(toolCall.Function.Arguments);
                    var resultJson = JsonSerializer.Serialize(result);

                    // Add tool response to messages
                    messages.Add(new ChatMessage
                    {
                        Role = MessageRole.Tool,
                        Content = resultJson,
                        ToolCallId = toolCall.Id
                    });

                    Console.WriteLine($"\nTool {toolCall.Function.Name} executed successfully");
                }
                catch (Exception ex)
                {
                    messages.Add(new ChatMessage
                    {
                        Role = MessageRole.Tool,
                        Content = $"Error: {ex.Message}",
                        ToolCallId = toolCall.Id

                    Console.WriteLine($"\nTool {toolCall.Function.Name} failed: {ex.Message}");

            Model = "mistral-large-latest",
            Messages = messages,

            ToolChoice = ToolChoice.Auto,

    #region File System Operations

    private async Task<object> ReadTextFileAsync(string arguments)
    {
        var args = JsonSerializer.Deserialize<Dictionary<string, string>>(arguments);
        var filePath = args?["filePath"] ?? throw new ArgumentException("filePath is required");

        if (!File.Exists(filePath))
            throw new FileNotFoundException($"File not found: {filePath}");

        var content = await File.ReadAllTextAsync(filePath);
        return new FileReadResult { Content = content, FilePath = filePath };
    }

    private async Task<object> WriteTextFileAsync(string arguments)
    {
        var args = JsonSerializer.Deserialize<Dictionary<string, string>>(arguments);
        var filePath = args?["filePath"] ?? throw new ArgumentException("filePath is required");
        var content = args?["content"] ?? throw new ArgumentException("content is required");

        var directory = Path.GetDirectoryName(filePath);
        if (!string.IsNullOrEmpty(directory) && !Directory.Exists(directory))
        {
            Directory.CreateDirectory(directory);
        }

        await File.WriteAllTextAsync(filePath, content);
        return new FileWriteResult { Success = true, FilePath = filePath };
    }

    private async Task<object> ScanFolderAsync(string arguments)
    {
        var args = JsonSerializer.Deserialize<Dictionary<string, string>>(arguments);
        var folderPath = args?["folderPath"] ?? throw new ArgumentException("folderPath is required");

        if (!Directory.Exists(folderPath))
            throw new DirectoryNotFoundException($"Directory not found: {folderPath}");

        var files = new List<FileInfo>();
        var directories = new List<DirectoryInfo>();

        foreach (var file in Directory.GetFiles(folderPath))
        {
            var fileInfo = new System.IO.FileInfo(file);
            files.Add(new FileInfo
            {
                Name = fileInfo.Name,
                Path = fileInfo.FullName,
                Size = fileInfo.Length,
                LastModified = fileInfo.LastWriteTime
            });
        }

        foreach (var dir in Directory.GetDirectories(folderPath))
        {
            var dirInfo = new System.IO.DirectoryInfo(dir);
            directories.Add(new DirectoryInfo
            {
                Name = dirInfo.Name,
                Path = dirInfo.FullName,
                LastModified = dirInfo.LastWriteTime
            });
        }

        return new FolderScanResult
        {
            Path = folderPath,
            Files = files,
            Directories = directories
        };
    }

    private async Task<object> DeleteFileAsync(string arguments)
    {
        var args = JsonSerializer.Deserialize<Dictionary<string, string>>(arguments);
        var filePath = args?["filePath"] ?? throw new ArgumentException("filePath is required");

        if (!File.Exists(filePath))

        return new FileOperationResult { Success = true };

    }
```

```csharp
private async Task<(FileSystemNode, int, int, long)> BuildFileTreeAsync(string rootPath, ScanOptions options)
{
    var rootNode = new FileSystemNode
    {
        Name = Path.GetFileName(rootPath) ?? rootPath,
        Path = rootPath,
        IsDirectory = true,
        LastModified = Directory.GetLastWriteTime(rootPath),
        Depth = 0
    };

    var totalFiles = 0;
    var totalDirs = 1; // Count the root directory
    var totalSize = 0L;

    if (options.Mode == ScanMode.Flat)
    {
        // Simple flat scan
        await ProcessDirectoryFlatAsync(rootPath, rootNode, options, ref totalFiles, ref totalDirs, ref totalSize);
    }
    else if (options.Mode == ScanMode.BreadthFirst)
    {
        // Breadth-first scan
        await ProcessDirectoryBreadthFirstAsync(rootPath, rootNode, options, ref totalFiles, ref totalDirs, ref totalSize);
    }
    else // DepthFirst
    {
        // Depth-first scan
        await ProcessDirectoryDepthFirstAsync(rootPath, rootNode, options, ref totalFiles, ref totalDirs, ref totalSize);
    }

    return (rootNode, totalFiles, totalDirs, totalSize);
}

private async Task ProcessDirectoryFlatAsync(
    string currentPath,
    FileSystemNode currentNode,
    ScanOptions options,
    ref int totalFiles,
    ref int totalDirs,
    ref long totalSize)
{
    try
    {
        if (options.IncludeDirectories)
        {
            foreach (var dir in Directory.GetDirectories(currentPath))
            {
                if (ShouldExclude(dir, options.ExcludePatterns)) continue;

                var dirInfo = new System.IO.DirectoryInfo(dir);
                var dirNode = new FileSystemNode
                {
                    Name = dirInfo.Name,
                    Path = dirInfo.FullName,
                    IsDirectory = true,
                    LastModified = dirInfo.LastWriteTime,
                    Depth = currentNode.Depth + 1
                };

                if (currentNode.Children == null)
                    currentNode.Children = new List<FileSystemNode>();
                currentNode.Children.Add(dirNode);
                totalDirs++;
            }
        }

        if (options.IncludeFiles)
        {
            foreach (var file in Directory.GetFiles(currentPath))
            {
                if (!MatchesPatterns(file, options.FilePatterns)) continue;

                var fileInfo = new System.IO.FileInfo(file);
                var fileNode = new FileSystemNode
                {
                    Name = fileInfo.Name,
                    Path = fileInfo.FullName,
                    IsDirectory = false,
                    Size = fileInfo.Length,
                    LastModified = fileInfo.LastWriteTime,
                    Depth = currentNode.Depth + 1
                };

                if (currentNode.Children == null)
                    currentNode.Children = new List<FileSystemNode>();
                currentNode.Children.Add(fileNode);
                totalFiles++;
                totalSize += fileInfo.Length;
            }
        }
    }
    catch (UnauthorizedAccessException)
    {
        // Skip directories we don't have access to
    }
}

private async Task ProcessDirectoryBreadthFirstAsync(
    string rootPath,
    FileSystemNode rootNode,
    ScanOptions options,
    ref int totalFiles,
    ref int totalDirs,
    ref long totalSize)
{
    var queue = new Queue<(string Path, FileSystemNode Node, int CurrentDepth)>();
    queue.Enqueue((rootPath, rootNode, 0));

    while (queue.Count > 0)
    {
        var (currentPath, currentNode, currentDepth) = queue.Dequeue();

        if (currentDepth >= options.MaxDepth) continue;

        try
        {
            if (options.IncludeDirectories)
            {
                foreach (var dir in Directory.GetDirectories(currentPath))
                {
                    if (ShouldExclude(dir, options.ExcludePatterns)) continue;

                    var dirInfo = new System.IO.DirectoryInfo(dir);
                    var dirNode = new FileSystemNode
                    {
                        Name = dirInfo.Name,
                        Path = dirInfo.FullName,
                        IsDirectory = true,
                        LastModified = dirInfo.LastWriteTime,
                        Depth = currentDepth + 1
                    };

                    if (currentNode.Children == null)
                        currentNode.Children = new List<FileSystemNode>();
                    currentNode.Children.Add(dirNode);
                    totalDirs++;

                    queue.Enqueue((dirInfo.FullName, dirNode, currentDepth + 1));
                }
            }

            if (options.IncludeFiles)
            {
                foreach (var file in Directory.GetFiles(currentPath))

private bool ShouldExclude(string path, string[]? excludePatterns)
{
    if (excludePatterns == null || excludePatterns.Length == 0)
        return false;

    var pathParts = path.Split(Path.DirectorySeparatorChar, StringSplitOptions.RemoveEmptyEntries);
    foreach (var pattern in excludePatterns)
    {
        if (pathParts.Any(p => p.Equals(pattern, StringComparison.OrdinalIgnoreCase)))
        {
            return true;
        }
    }
    return false;
}

private bool MatchesPatterns(string filePath, string[]? filePatterns)
{
    if (filePatterns == null || filePatterns.Length == 0)
        return true;

    var fileName = Path.GetFileName(filePath);
    foreach (var pattern in filePatterns)
    {
        if (pattern == "*") return true;

        try
        {
            // Simple pattern matching (for *.ext patterns)
            if (pattern.StartsWith("*."))
            {
                var ext = pattern[1..];
                if (fileName.EndsWith(ext, StringComparison.OrdinalIgnoreCase))
                    return true;
            }
            else if (fileName.Equals(pattern, StringComparison.OrdinalIgnoreCase))
            {
                return true;
            }
        }
        catch
        {
            // If pattern matching fails, include the file
        }
    }
    return false;
}
```

```csharp
    foreach (var pattern in filePatterns)
    {
        try
        {
            if (pattern == "*") return true;

            // Handle extension patterns (e.g., "*.txt")
            if (pattern.StartsWith("*."))
            {
                var patternExt = pattern[2..]; // Remove "*." prefix
                if (ext.Equals(patternExt, options.CaseSensitive
                    ? StringComparison.Ordinal
                    : StringComparison.OrdinalIgnoreCase))
                {
                    return true;
                }
            }
            // Handle full file name patterns
            else if (fileName.Equals(pattern, options.CaseSensitive
                ? StringComparison.Ordinal
                : StringComparison.OrdinalIgnoreCase))
            {
                return true;
            }
            // Handle patterns with wildcards (simplified)
            else if (pattern.Contains('*'))
            {
                // Convert to regex pattern
                var regexPattern = "^" + Regex.Escape(pattern).Replace("\\*", ".*") + "$";
                if (Regex.IsMatch(fileName, regexPattern, options.CaseSensitive
                    ? RegexOptions.None
                    : RegexOptions.IgnoreCase))
                {
                    return true;
                }
            }
        }
        catch
        {
            // If pattern matching fails, include the file
        }
    }
    return false;
}
```

```csharp
private async Task ProcessDirectoryBreadthFirstAsync(
    string rootPath,
    FileSystemNode rootNode,
    ScanOptions options,
    ref int totalFiles,
    ref int totalDirs,
    ref long totalSize)
{
    var queue = new Queue<(string Path, FileSystemNode Node, int CurrentDepth)>();
    queue.Enqueue((rootPath, rootNode, 0));

    while (queue.Count > 0)
    {
        var (currentPath, currentNode, currentDepth) = queue.Dequeue();

        if (currentDepth >= options.MaxDepth) continue;

        try
        {
            if (options.IncludeDirectories)
            {
                foreach (var dir in Directory.GetDirectories(currentPath))
                {
                    if (ShouldIgnore(dir, true, options)) continue;

                    var dirInfo = new System.IO.DirectoryInfo(dir);
                    var dirNode = new FileSystemNode
                    {
                        Name = dirInfo.Name,
                        Path = dirInfo.FullName,
                        IsDirectory = true,
                        LastModified = dirInfo.LastWriteTime,
                        Depth = currentDepth + 1
                    };

                    if (currentNode.Children == null)
                        currentNode.Children = new List<FileSystemNode>();
                    currentNode.Children.Add(dirNode);
                    totalDirs++;

                    queue.Enqueue((dirInfo.FullName, dirNode, currentDepth + 1));
                }
            }

            if (options.IncludeFiles)
            {
                foreach (var file in Directory.GetFiles(currentPath))
                {
                    if (ShouldIgnore(file, false, options)) continue;
                    if (!MatchesPatterns(file, options.FilePatterns, options)) continue;

                    var fileInfo = new System.IO.FileInfo(file);
                    var fileNode = new FileSystemNode
                    {
                        Name = fileInfo.Name,
                        Path = fileInfo.FullName,
                        IsDirectory = false,
                        Size = fileInfo.Length,
                        LastModified = fileInfo.LastWriteTime,
                        Depth = currentDepth + 1
                    };

                    if (currentNode.Children == null)
                        currentNode.Children = new List<FileSystemNode>();
                    currentNode.Children.Add(fileNode);
                    totalFiles++;
                    totalSize += fileInfo.Length;

            }

        }
    }
```

```csharp
private async Task<object> ScanFolderTreeAsync(string arguments)
{
    var args = JsonSerializer.Deserialize<Dictionary<string, object>>(arguments);
    var folderPath = args?["folderPath"] as string ?? throw new ArgumentException("folderPath is required");

    // Parse options with defaults
    var options = new ScanOptions();

    // Try to load default ignore lists from file
    try
    {
        if (File.Exists("scan_ignore.json"))
        {
            var ignoreJson = await File.ReadAllTextAsync("scan_ignore.json");
            var ignoreConfig = JsonSerializer.Deserialize<ScanOptions>(ignoreJson);

            if (ignoreConfig != null)
            {
                options.IgnoreFolders = ignoreConfig.IgnoreFolders ?? options.IgnoreFolders;
                options.IgnoreFiles = ignoreConfig.IgnoreFiles ?? options.IgnoreFiles;
                options.IgnoreExtensions = ignoreConfig.IgnoreExtensions ?? options.IgnoreExtensions;
            }
        }
    }
    catch
    {
        // If we can't load the ignore file, just continue with defaults
    }

    // Override with any options from the arguments
    if (args != null)
    {
        if (args.TryGetValue("mode", out var modeObj) && modeObj is string modeStr)
        {
            Enum.TryParse(modeStr, true, out options.Mode);
        }
        if (args.TryGetValue("maxDepth", out var depthObj) && depthObj is JsonElement depthElement)
        {
            options.MaxDepth = depthElement.GetInt32();
        }
        if (args.TryGetValue("filePatterns", out var patternsObj) && patternsObj is JsonElement patternsElement)
        {
            options.FilePatterns = patternsElement.EnumerateArray()
                .Select(e => e.GetString() ?? "")
                .Where(s => !string.IsNullOrEmpty(s))
                .ToArray();
        }
        if (args.TryGetValue("ignoreFolders", out var ignoreFoldersObj) && ignoreFoldersObj is JsonElement ignoreFoldersElement)
        {
            var ignoreFolders = ignoreFoldersElement.EnumerateArray()
                .Select(e => e.GetString() ?? "")
                .Where(s => !string.IsNullOrEmpty(s))
                .ToArray();
            options.IgnoreFolders = options.IgnoreFolders?.Concat(ignoreFolders).Distinct().ToArray() ?? ignoreFolders;
        }
        if (args.TryGetValue("ignoreFiles", out var ignoreFilesObj) && ignoreFilesObj is JsonElement ignoreFilesElement)
        {
            var ignoreFiles = ignoreFilesElement.EnumerateArray()
                .Select(e => e.GetString() ?? "")
                .Where(s => !string.IsNullOrEmpty(s))
                .ToArray();
            options.IgnoreFiles = options.IgnoreFiles?.Concat(ignoreFiles).Distinct().ToArray() ?? ignoreFiles;
        }
        if (args.TryGetValue("ignoreExtensions", out var ignoreExtObj) && ignoreExtObj is JsonElement ignoreExtElement)
        {
            var ignoreExtensions = ignoreExtElement.EnumerateArray()
                .Select(e => e.GetString() ?? "")
                .Where(s => !string.IsNullOrEmpty(s))
                .ToArray();
            options.IgnoreExtensions = options.IgnoreExtensions?.Concat(ignoreExtensions).Distinct().ToArray() ?? ignoreExtensions;
        }
        if (args.TryGetValue("caseSensitive", out var caseSensitiveObj) && caseSensitiveObj is JsonElement caseElement)
        {
            options.CaseSensitive = caseElement.GetBoolean();
        }
    }

    if (!Directory.Exists(folderPath))
        throw new DirectoryNotFoundException($"Directory not found: {folderPath}");

    var (rootNode, totalFiles, totalDirs, totalSize) = await BuildFileTreeAsync(folderPath, options);

    return new FileTreeScanResult
    {
        RootPath = folderPath,
        Mode = options.Mode,
        MaxDepth = options.MaxDepth,
        RootNode = rootNode,
        TotalFiles = totalFiles,
        TotalDirectories = totalDirs,
        TotalSize = totalSize
    };
}
```

```csharp
private async Task HandleCommandAsync(string command)
{
    var parts = command.Trim().Split(' ', 2);
    var cmd = parts[0].ToLower();
    var arg = parts.Length > 1 ? parts[1] : string.Empty;

private void LoadSkill(Skill skill)
{
    // Store the skill
    _loadedSkills[skill.Name] = skill;

    // Apply skill configuration
    if (skill.Temperature.HasValue)
    {
        _temperature = skill.Temperature.Value;
    }

    if (!string.IsNullOrEmpty(skill.SystemPrompt))
    {
        // Update or add system prompt
        var systemMessage = _messages.FirstOrDefault(m => m.Role == MessageRole.System);
        if (systemMessage != null)
        {
            systemMessage.Content = $"{systemMessage.Content}\n\n{skill.SystemPrompt}";
        }
        else
        {
            _messages.Insert(0, new ChatMessage
            {
                Role = MessageRole.System,
                Content = $"{_systemPrompt}\n\n{skill.SystemPrompt}"
            });
        }
    }

    // Add skill messages to context
    if (skill.Messages != null)
    {
        foreach (var message in skill.Messages)
        {
            if (!_messages.Any(m => m.Role == message.Role && m.Content == message.Content))
            {
                _messages.Add(message);
            }
        }
    }

    // Register skill tools
    if (skill.Tools != null)
    {
        foreach (var tool in skill.Tools)
        {
            if (tool.Function != null && !string.IsNullOrEmpty(tool.Function.Name))
            {
                // Check if this tool is already registered
                if (!_toolFunctions.ContainsKey(tool.Function.Name))
                {
                    // For now, we'll just add a placeholder. In a real implementation,
                    // you'd want to dynamically register the actual function.
                    _toolFunctions[tool.Function.Name] = async (args) =>
                    {
                        return new { error = $"Tool {tool.Function.Name} is not implemented" };
                    };
                }
            }
        }
    }
}

private void UnloadSkill(string skillName)
{
    if (!_loadedSkills.TryGetValue(skillName, out var skill))
        return;

    // Remove skill messages from context
    if (skill.Messages != null)
    {
        foreach (var message in skill.Messages)
        {
            _messages.RemoveAll(m =>
                m.Role == message.Role &&
                m.Content == message.Content &&
                m.ToolCalls == message.ToolCalls &&
                m.ToolCallId == message.ToolCallId);
        }
    }

    // Remove skill system prompt if it was added
    if (!string.IsNullOrEmpty(skill.SystemPrompt))
    {
        var systemMessage = _messages.FirstOrDefault(m => m.Role == MessageRole.System);
        if (systemMessage != null)
        {
            systemMessage.Content = systemMessage.Content.Replace(skill.SystemPrompt, "").Trim();
        }
    }

    // Remove skill tools
    if (skill.Tools != null)
    {
        foreach (var tool in skill.Tools)
        {
            if (tool.Function != null)
            {
                _toolFunctions.Remove(tool.Function.Name);
            }
        }
    }

    // Remove from loaded skills
    _loadedSkills.Remove(skillName);
}

private async Task LoadSkillsManifestAsync()
{
    var manifestPath = Path.Combine(_skillsDirectory, _skillsManifestFile);
    if (!File.Exists(manifestPath))
        return;

    try
    {
        var json = await File.ReadAllTextAsync(manifestPath);
        var manifest = JsonSerializer.Deserialize<SkillManifest>(json);

        if (manifest?.DefaultSkills != null)
        {
            foreach (var skillName in manifest.DefaultSkills)
            {
                var skillFile = Path.Combine(_skillsDirectory, $"{skillName}.json");
                if (File.Exists(skillFile))
                {
                    var skill = await LoadSkillFromFileAsync(skillFile);
                    if (skill != null)
                    {
                        LoadSkill(skill);
                        Console.WriteLine($"Loaded default skill: {skill.Name}");
                    }
                }
            }
        }
    }
    catch (Exception ex)
    {
        Console.WriteLine($"Warning: Could not load skills manifest - {ex.Message}");
    }
}
```

### Key Features Added:

1. **Skill Management System**:
   - `/skill {filename}`: Toggles loading/unloading of a skill
     - First call loads the skill
     - Second call unloads the skill
   - `/skills`: Lists all currently loaded skills
   - Skills are stored in a `skills` directory
   - Each skill is a JSON file with:
     - Name and description
     - System prompt additions
     - Pre-defined messages to add to context
     - Temperature setting
     - Additional tools

2. **Skill Manifest**:
   - `skills.json` file defines available skills
   - Can specify default skills to load on startup
   - Each skill references its JSON file

3. **Skill Loading/Unloading**:
   - When loaded, a skill:
     - Adds its system prompt to the context
     - Adds its messages to the conversation
     - Registers its tools
     - Can override temperature setting
   - When unloaded, all skill-specific additions are removed

4. **Context Integration**:
   - Skills modify the conversation context
   - Changes are persistent while the skill is loaded
   - Multiple skills can be loaded simultaneously
